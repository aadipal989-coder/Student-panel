function updateFeeStatus(e) { e.preventDefault(); // Form se values lena const studentEmail = e.target.querySelector('input[type="email"]')?.value || 'student@campus.com'; const totalFees = e.target.querySelectorAll('input[type="number"]')[0]?.value || 0; const paidAmount = e.target.querySelectorAll('input[type="number"]')[1]?.value || 0; const studentFeeData = { studentEmail: studentEmail, totalFees: Number(totalFees), paidAmount: Number(paidAmount) }; // Live Backend API par data bhejna fetch('https://campusflow-backend-wg92.onrender.com/api/admin/update-fee', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(studentFeeData) }) .then(res => res.json()) .then(data => { if(data.success) { alert("✅ Backend & MongoDB me Fee update ho gayi!"); switchTab('recordsTab', document.querySelectorAll('.tab-btn')[1]); } else { alert("❌ Fee Update Failed: " + data.message); } }) .catch(err => { console.error("Error:", err); alert("❌ Server se connect nahi ho paya!"); }); }
